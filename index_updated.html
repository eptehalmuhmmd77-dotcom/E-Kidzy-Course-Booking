واتساب الفورم (اختياري)
رقم واتساب يظهر للعميل في آخر خطوة الحجز علشان يتواصل معاك مباشرة. لو مش عايزه اتركه فاضي.
📖 طريقة ربط Google Sheets:
1 افتح Google Sheets وأنشئ ملف جديد
2 من القائمة: Extensions → Apps Script
3 امسح الكود الموجود والصق الكود ده:
function doPost(e) {
  const ss   = SpreadsheetApp.getActiveSpreadsheet();
  const data = JSON.parse(e.postData.contents);

  // اختار الشيت حسب المصدر — أو اعمل واحد لو مش موجود
  const sheetName = data.source === 'مسبق'       ? 'Advanced'
                  : data.source === 'انتظار عام' ? 'General'
                  : data.source === 'تحويل'      ? 'Transfers'
                  : data.source === 'test'        ? 'Test'
                  : 'Immediate';

  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) sheet = ss.insertSheet(sheetName);

  // هيدر تلقائي لو الشيت فاضي
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'المصدر','الاسم','الاسم بالإنجليزي','الموبايل','العمر',
      'المستوى','اليوم','الوقت','التيتشر','الشهر',
      'نوع الدفع','المبلغ','الحالة','التاريخ','ID'
    ]);
    sheet.getRange(1,1,1,15).setFontWeight('bold').setBackground('#4A154B').setFontColor('#FFFFFF');
  }

  sheet.appendRow([
    data.source   || '',
    data.name     || '',
    data.nameEn   || '',
    data.phone    || '',
    data.age      || '',
    data.level    || '',
    data.day      || '',
    data.time     || '',
    data.teacher  || '',
    data.month    || '',
    data.payType  || '',
    data.amount   || 0,
    data.status   || '',
    data.timestamp|| '',
    data.id       || ''
  ]);

  return ContentService.createTextOutput('OK');
}
1 افتح Google Sheets وأنشئ ملف جديد
2 من القائمة: Extensions → Apps Script
3 امسح الكود الموجود والصق الكود اللي فوق ☝️
4 اضغط Deploy → New deployment → Web app
5 اختار: Execute as: Me و Who has access: Anyone
6 اضغط Deploy وانسخ الـ Web app URL والصقه فوق 👆
7 اضغط 💾 Save URL ثم 🧪 Test Connection وتحقق من الشيت — هتلاقي row جديد في شيت Test
8 اضغط 🔄 Sync All Data لرفع كل بيانات قوائم الانتظار — هيتقسموا تلقائياً على شيتات منفصلة (Immediate / Advanced / General)
